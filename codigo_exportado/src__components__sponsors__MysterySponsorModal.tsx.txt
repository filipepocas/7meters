/**
 * 7meters - Mystery Sponsor Modal Component
 * Interface para seleção, revelação de valores mistério e assinatura
 * de patrocínios para o clube.
 */

import React, { useState } from 'react';
import { useGameStore } from '../../store/useGameStore';
import { Sponsor } from '../../types/sponsor.types';
import { generateMysterySponsorCard } from '../../utils/generators/sponsorGen';

interface MysterySponsorModalProps {
  isOpen: boolean;
  onClose: () => void;
}

export const MysterySponsorModal: React.FC<MysterySponsorModalProps> = ({ isOpen, onClose }) => {
  const { userClub, signMysterySponsor } = useGameStore();
  const [availableSponsors] = useState<Sponsor[]>(() => [
    generateMysterySponsorCard('Nacional'),
    generateMysterySponsorCard('Internacional'),
    generateMysterySponsorCard('Cripto / Tech'),
  ]);

  const [selectedSponsor, setSelectedSponsor] = useState<Sponsor | null>(null);
  const [revealedBonus, setRevealedBonus] = useState<number | null>(null);
  const [statusMessage, setStatusMessage] = useState<string | null>(null);

  if (!isOpen || !userClub) return null;

  const handleSelectSponsor = (sponsor: Sponsor) => {
    setSelectedSponsor(sponsor);
    setRevealedBonus(null);
    setStatusMessage(null);
  };

  const handleSignContract = () => {
    if (!selectedSponsor) return;

    const { injectedAmount } = signMysterySponsor(selectedSponsor);
    setRevealedBonus(injectedAmount);
    setStatusMessage(`Contrato assinado! Injeção imediata de €${injectedAmount.toLocaleString()} nas finanças do clube.`);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 p-4">
      <div className="w-full max-w-3xl rounded-none border-4 border-black bg-white p-6 shadow-[8px_8px_0px_0px_rgba(0,0,0,1)] dark:bg-zinc-900 dark:border-white dark:text-white">
        
        {/* Cabeçalho */}
        <div className="flex items-center justify-between border-b-4 border-black pb-4 dark:border-white">
          <h2 className="text-2xl font-black uppercase tracking-wider">
            🎁 Caixas de Patrocínio Mistério
          </h2>
          <button
            onClick={onClose}
            className="border-2 border-black bg-red-500 px-3 py-1 font-bold text-white hover:bg-red-600 dark:border-white"
          >
            X
          </button>
        </div>

        <p className="my-3 text-sm font-bold text-zinc-600 dark:text-zinc-300">
          Escolha uma das propostas de patrocínio cegas. O valor base é garantido, mas o prémio de assinatura será revelado após a confirmação do contrato.
        </p>

        {/* Lista de Patrocinadores Disponíveis */}
        <div className="my-4 grid grid-cols-1 gap-4 md:grid-cols-3">
          {availableSponsors.map((sponsor) => {
            const isSelected = selectedSponsor?.id === sponsor.id;
            return (
              <div
                key={sponsor.id}
                onClick={() => handleSelectSponsor(sponsor)}
                className={`cursor-pointer border-4 p-4 transition-all ${
                  isSelected
                    ? 'border-yellow-400 bg-yellow-100 shadow-[6px_6px_0px_0px_rgba(0,0,0,1)] dark:bg-yellow-950 dark:text-white'
                    : 'border-black bg-zinc-50 hover:bg-zinc-100 dark:border-white dark:bg-zinc-800 dark:hover:bg-zinc-700'
                }`}
              >
                <div className="text-xs font-black uppercase text-zinc-500 dark:text-zinc-400">
                  {sponsor.tier}
                </div>
                <div className="text-lg font-black">{sponsor.companyName}</div>
                
                <div className="mt-3 space-y-1 font-mono text-xs font-bold">
                  <div>Base Semanal: €{sponsor.weeklyAmount.toLocaleString()}</div>
                  <div>Duração: {sponsor.contractWeeks} Semanas</div>
                  <div className="text-blue-600 dark:text-blue-400">
                    Bónus Revelável: ??? €
                  </div>
                </div>
              </div>
            );
          })}
        </div>

        {/* Painel de Ação e Assinatura */}
        {selectedSponsor && !revealedBonus && (
          <div className="mt-6 border-4 border-black bg-zinc-100 p-4 dark:border-white dark:bg-zinc-800">
            <h3 className="text-lg font-black uppercase mb-2">
              Confirmar Contrato: {selectedSponsor.companyName}
            </h3>
            <p className="text-xs font-bold text-zinc-600 dark:text-zinc-300 mb-4">
              Ao assinar este contrato, o teu clube compromete-se com a marca por {selectedSponsor.contractWeeks} semanas e revela o prémio de entrada mistério.
            </p>

            <button
              onClick={handleSignContract}
              className="w-full border-2 border-black bg-green-500 p-3 font-black uppercase text-white hover:bg-green-600 shadow-[4px_4px_0px_0px_rgba(0,0,0,1)]"
            >
              ✍️ Assinar Contrato e Revelar Bónus Mistério
            </button>
          </div>
        )}

        {/* Resultado do Bónus Revelado */}
        {statusMessage && (
          <div className="mt-6 border-4 border-black bg-green-200 p-4 font-black text-green-900 shadow-[4px_4px_0px_0px_rgba(0,0,0,1)] dark:bg-green-950 dark:text-green-100 dark:border-white">
            🎉 {statusMessage}
          </div>
        )}
      </div>
    </div>
  );
};
